## 실제로 완료한 세트에서 시트에 남길 값을 구합니다

`supabase/functions/_shared/routineSheet.ts`는 시트의 일차별 계획과 회차별 기록을 읽고, 앱 기록을 시트에 반영할 값을 구성합니다. 아래 함수는 완료한 세트만 대상으로 중량을 정리합니다. 숫자와 기호가 섞인 표현도 형태를 보존합니다.

```typescript
export function sheetCellFor(sets: { kg: string | null; done: boolean }[]): string {
  const done = sets.filter((s) => s.done);
  if (!done.length) return '';
  const kgs = done.map((s) => (s.kg ?? '').trim()).filter(Boolean);
  if (!kgs.length) return '-';
  const shape = weightShape(kgs[0]);
  if (shape.includes('#') && kgs.every((k) => weightShape(k) === shape)) {
    const parts = kgs.map((k) => k.match(/\d+(?:\.\d+)?/g) ?? []);
    const best = parts[0].map((_, i) => parts.reduce((m, p) => (Number(p[i]) > Number(m) ? p[i] : m), parts[0][i]));
    let i = 0;
    return canonCell(kgs[0].replace(/\d+(?:\.\d+)?/g, () => best[i++]).slice(0, 50));
  }
  let best = kgs[0];
  for (const k of kgs.slice(1)) if (heavier(weightKey(k), weightKey(best))) best = k;
  best = canonCell(best);
  return best.slice(0, 50);
}
```

## 같은 운동의 회원 메모와 트레이너 설명을 나눕니다

`src/lib/exercise.ts`에서는 운동을 식별한 뒤 작성자에 따라 두 메모를 구분합니다. 루틴이 달라져도 같은 운동의 메모를 참고하는 구조입니다.

```typescript
export function memoPair(memos: ExerciseMemo[] | null | undefined, name: string): MemoPair {
  const k = exerciseKey(name);
  const mine = (memos ?? []).filter((m) => m.key === k);
  return { member: mine.find((m) => m.author === 'member') ?? null, trainer: mine.find((m) => m.author === 'trainer') ?? null };
}
```

## 값이 없는 항목을 변화가 없는 것으로 처리하지 않습니다

`src/lib/report.ts`는 비교 가능한 측정값만 설명하고, 한쪽에 값이 없으면 비교할 수 없는 항목으로 표시합니다. 기간별 보고서에서도 이 비교 기준을 사용합니다.

```typescript
export function changeSentences(c: Change): string {
  const out: string[] = [];
  const missing: string[] = [];
  if (c.pbf === null) missing.push('체지방률');
  else if (c.pbf < 0) out.push(`체지방률이 ${Math.abs(c.pbf).toFixed(1)}%p 내려갔어요.`);
  else if (c.pbf > 0) out.push(`체지방률은 ${c.pbf.toFixed(1)}%p 늘었어요.`);
  else out.push('체지방률은 그대로예요.');
  if (c.smm === null) missing.push('골격근량');
  else if (c.smm > 0) out.push(`골격근량이 ${c.smm.toFixed(1)}kg 늘었어요.`);
  else if (c.smm < 0) out.push(`골격근량은 ${Math.abs(c.smm).toFixed(1)}kg 줄었어요.`);
  else out.push('골격근량은 그대로예요.');
  if (missing.length === 2) out.push('골격근량·체지방률도 기록하면 변화를 함께 볼 수 있어요.');
  else if (missing.length === 1) out.push(`${missing[0]}은 기록이 없어 비교할 수 없어요.`);
  return out.join(' ');
}
```
